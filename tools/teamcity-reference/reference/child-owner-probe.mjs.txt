async function simultaneous(label, commands, match) {
  const latch = Promise.withResolvers();
  const timer = setTimeout(() => latch.reject(new Error('Both clients did not reach dispatch barrier.')), 10000);
  barrier = { match, ready: 0, promise: latch.promise, release: latch.resolve };
  try {
    const results = await Promise.allSettled([invoke(commands[0], a, label + '-A'), invoke(commands[1], b, label + '-B')]);
    receipt.observations.push({ label, results: results.map(r => r.status === 'fulfilled' ? { status: r.status, value: r.value } : { status: r.status, error: String(r.reason) }) });
    if (results.some(r => r.status !== 'fulfilled')) throw new Error('Concurrent request failed; retain as diagnostic, not accepted reference.');
  } finally { clearTimeout(timer); barrier = undefined; }
}
